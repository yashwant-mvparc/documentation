---
title: 'List Contacts'
description: 'Fetch all profiles with their segmentation and health status. Supports filtering by specific list segments.'
api: 'GET /subscribers'
---

### Query Parameters

<ParamField query="page" type="integer">
  Page number. Defaults to `1`.
</ParamField>

<ParamField query="limit" type="integer">
  Number of records per page. Defaults to `20`.
</ParamField>

<ParamField query="search" type="string">
  Filter by name, email, or company.
</ParamField>

<ParamField query="company" type="string">
  Filter contacts by company name (case-insensitive substring match or employer association).
</ParamField>

<ParamField query="company_id" type="string">
  Filter contacts linked to a specific company record UUID.
</ParamField>

<ParamField query="email" type="string">
  Filter contacts matching an email address.
</ParamField>

<ParamField query="phone" type="string">
  Filter contacts matching a phone number.
</ParamField>

<ParamField query="tags" type="string">
  Filter contacts by tag (comma-separated string).
</ParamField>

<ParamField query="is_subscribed" type="boolean">
  Filter contacts by email subscription status (`true` or `false`).
</ParamField>

<ParamField query="recipient_list_id" type="string">
  Filter contacts belonging to a specific segment or list ID.
</ParamField>
